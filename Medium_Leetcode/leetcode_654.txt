/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    int maxIndex(vector<int> &nums, int start, int end) {
        int mxVal=INT_MIN; int idx=-1;
        for (int i=start; i<=end; i++) {
            if (nums[i]>mxVal) {
                mxVal=nums[i];
                idx=i;
            }
        }
        return idx;
    }
    TreeNode* constructMaximumBinaryTree(vector<int>& nums) {
        if (nums.empty()) return nullptr;
        int index=maxIndex(nums, 0, nums.size()-1);
        TreeNode *temp=new TreeNode(nums[index]);
        vector<int> leftNums(nums.begin(), nums.begin() + index);
        vector<int> rightNums(nums.begin() + index + 1, nums.end());
        temp->left=constructMaximumBinaryTree(leftNums);
        temp->right=constructMaximumBinaryTree(rightNums);
        return temp;
    }
};
