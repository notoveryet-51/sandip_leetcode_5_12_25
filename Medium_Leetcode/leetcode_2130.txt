/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    int pairSum(ListNode* head) {
        vector<int> result;
        ListNode *temp=head;
        while (temp) {
            result.push_back(temp->val);
            temp=temp->next;
        }
        int maxSum=INT_MIN;
        int n=result.size();
        for (int i=0; i<n/2; i++) {
            maxSum=max(maxSum, result[i]+result[n-1-i]);
        }
        return maxSum;
    }
};
