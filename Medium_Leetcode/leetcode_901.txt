class StockSpanner {
public:
    vector<int> vec;
    StockSpanner() {
        
    }
    
    int next(int price) {
        vec.push_back(price);
        int n=vec.size();
        int count=1;
        int comp=price;
        for (int i=n-2; i>=0; i--) {
            if (vec[i]<=price) count++;
            else break;
        }
        return count;
    }
};

 * Your StockSpanner object will be instantiated and called as such:
 * StockSpanner* obj = new StockSpanner();
 * int param_1 = obj->next(price);
