class Solution {
public:
    int nearestExit(vector<vector<char>>& maze, vector<int>& entrance) {
        int m=maze.size();
        int n=maze[0].size();
        queue<pair<pair<int, int>, int>> q;
        q.push({{entrance[0], entrance[1]}, 0});
        maze[entrance[0]][entrance[1]]='#';
        int minDist=INT_MAX;

        // four way movement
        vector<int> delRow={0, -1, 1, 0};
        vector<int> delCol={1, 0, 0, -1};

        // start the movement
        while (!q.empty())  {
            auto it=q.front();
            int row=it.first.first;
            int col=it.first.second;
            int dist=it.second;
            q.pop();
            for (int i=0; i<4; i++) {
                int currRow=row+delRow[i];
                int currCol=col+delCol[i];
                if (currRow>=0 && currRow<m && currCol>=0 && currCol<n && maze[currRow][currCol]=='.') {
                    maze[currRow][currCol]='#';
                    if (currRow==0 || currCol==0 || currCol==n-1 || currRow==m-1) {
                        minDist=min(minDist, dist+1); continue;
                    }
                    q.push({{currRow, currCol}, dist+1});
                }
            }
        }
        return minDist==INT_MAX ? -1 : minDist;
    }
};
