Leetcode 740

class Solution {
public:
    int solve(int i, int maxVal, vector<int>& total, vector<int>& dp) {
        if (i > maxVal) return 0;
        if (dp[i] != -1) return dp[i];
        int take = total[i] + solve(i + 2, maxVal, total, dp); // we should skip nums[i]+1
        int notTake = solve(i + 1, maxVal, total, dp);
        return dp[i] = max(take, notTake);
    }

    int deleteAndEarn(vector<int>& nums) {
        int maxVal = *max_element(nums.begin(), nums.end());
        vector<int> total(maxVal + 1, 0);
        for (int x : nums) { // here we form the unique total array. [2,2,3,3,3,4] gets converted to [0, 0, 4, 9, 4] where the indices represent the nums[i]
            total[x] += x;
        }
        vector<int> dp(maxVal + 1, -1);
        return solve(1, maxVal, total, dp);
    }
};
