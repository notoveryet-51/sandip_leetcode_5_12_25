class Solution {
public:
    vector<string> findRepeatedDnaSequences(string s) {
        int n=s.size();
        unordered_map<string, int> mpp;
        for (int i=0; i<=n-9; i++) {
            mpp[s.substr(i, 10)]++;
        }
        vector<string>result;
        if (mpp.empty()) return {};
        for (auto it: mpp) {
            if (it.second>1) result.push_back(it.first);
        }
        return result;
    }
};
