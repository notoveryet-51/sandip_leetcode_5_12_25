class Solution {
    public String evaluate(String s, List<List<String>> knowledge) {
        int n=s.length();
        int m=knowledge.size();
        StringBuilder sb=new StringBuilder();
        StringBuilder s1=new StringBuilder();
        StringBuilder res=new StringBuilder();
        Map<String, String> mpp=new HashMap<>();
        for (int i=0; i<m; i++) {
            mpp.put(knowledge.get(i).get(0), knowledge.get(i).get(1));
        }
        for (int i=0; i<n; ) {
            if (s.charAt(i)=='(') {
                sb.setLength(0);
                int j=i+1;
                for (; j<n; j++) {
                    if (s.charAt(j)==')') break;
                    sb.append(s.charAt(j));
                }
                String temp=mpp.getOrDefault(sb.toString(), "?");
                res.append(temp); 
                i=j+1; 
            } else {
                res.append(s.charAt(i)); 
                i++; 
            }
        }
        return res.toString();
    }
}
