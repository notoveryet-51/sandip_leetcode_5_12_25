class Solution {
public:
    vector<int> executeInstructions(int n, vector<int>& startPos, string s) {
        int len=s.size();
        vector<int> moves;
        for (int i=0; i<len; i++) {
            int count=0;
            int x=startPos[0], y=startPos[1];
            for (int j=i; j<len; j++) {
                if (s[j]=='R') y+=1;
                else if (s[j]=='U') x-=1;
                else if (s[j]=='D') x+=1;
                else y-=1;
                if (x==n || y==n || x<0 || y<0) break;
                else count++;
            }
            moves.push_back(count);
        }
        return moves;
    }
};
