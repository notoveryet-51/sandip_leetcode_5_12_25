class Solution {
public:
    int maxSubarrayLength(vector<int>& nums, int k) {
        int i=0, j=0;
        int mxLen=0;
        unordered_map<int, int> mp;
        while (j<nums.size()) {
            mp[nums[j]]++;
            while (mp[nums[j]]>k) {
                mp[nums[i]]--;
                if (mp[nums[i]]==0) mp.erase(nums[i]);
                i++;
            }
            mxLen=max(mxLen, j-i+1);
            j++;
        }
        return mxLen;
    }
};
