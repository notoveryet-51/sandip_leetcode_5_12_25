class Solution {
public:
    long long minSumSquareDiff(vector<int>& nums1, vector<int>& nums2, int k1, int k2) {
        int n = nums1.size();
        int mx = 0;
        vector<int> diff(n);
        for (int i = 0; i < n; i++) {
            diff[i] = abs(nums1[i] - nums2[i]);
            mx = max(mx, diff[i]); // find the max difference
        }

        vector<long long> cnt(mx + 1, 0);
        for (int d : diff) cnt[d]++;

        long long ops = (long long)k1 + k2;

        for (int v = mx; v >= 1 && ops > 0; v--) {
            if (cnt[v] == 0) continue;
            long long move = min(cnt[v], ops);   // elements we can lower by 1
            cnt[v]     -= move;
            cnt[v - 1] += move;
            ops        -= move;
        }

        long long ans = 0;
        for (long long v = 1; v <= mx; v++) {
            ans += cnt[v] * v * v;
        }
        return ans;
    }
};
