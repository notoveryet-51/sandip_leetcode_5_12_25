class Solution {
public:
    int countDistinctIntegers(vector<int>& nums) {
        unordered_map<int, int> mpp;
        for (auto it: nums) {
            mpp[it]++;
        }
        for (int i=0; i<nums.size(); i++) {
            int copy=nums[i];
            int sum=0;
            while (copy>0) {
                int dig=copy%10;
                sum=sum*10+dig;
                copy/=10;
            }
            mpp[sum]++;
        }
        return mpp.size();
    }
};
