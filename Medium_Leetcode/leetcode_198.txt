class Solution {
public:
    int rob(vector<int>& nums) {
        int n=nums.size();
        if (n==1) return nums[0];
        vector<int> dp(n);
        dp[0]=nums[0];
        dp[1]=max(nums[0], nums[1]);
        for (int i=2; i<n; i++) {
            dp[i]=max(dp[i-1], nums[i]+ dp[i-2]);
        }
        return dp[n - 1];
    }
};

class Solution {
public:
    int n;
    int helperFunction (vector<int> &dp, vector<int> &nums, int i) {
        if (i>=n) return 0;
        if (dp[i]!=-1) return dp[i];
        int rob=helperFunction(dp, nums, i+2)+nums[i];
        int notRob=helperFunction(dp, nums, i+1);
        return dp[i]=max(rob, notRob);
    }
    int rob(vector<int>& nums) {
        n=nums.size();
        vector<int> dp(n+1, -1);
        return helperFunction(dp, nums, 0);
    }
};
