class Solution {
public:
    int minInsertions(string s) {
        int count=0;
        int finalCount=0;
        for (auto it: s) {
            if (it=='(') count+=2;
            else count--;
            if (count==-2) {
                count=0;
                finalCount++;
            }
        }
        if (count>0) finalCount+=count;
        else if (count<0) finalCount+=2;
        return finalCount;
    }
};
