class Solution {
public:
    int minDeletions(string s) {
        unordered_map<char, int> mpp;
        int n=s.length();
        for(auto it: s) mpp[it]++;
        vector<int> p;
        for (auto it: mpp) {
            p.push_back(it.second);
        }
        sort(p.rbegin(), p.rend());
        int count=0;
        for (int i=1; i<p.size(); i++) {
            while (p[i]!=0 && p[i]>=p[i-1]) {
                p[i]--; count++;
            }
        }
        return count;
    }
};
