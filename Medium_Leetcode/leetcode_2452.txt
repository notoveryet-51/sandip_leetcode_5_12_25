class Solution {
public:
    vector<string> twoEditWords(vector<string>& queries, vector<string>& dictionary) {
        vector<string> result;
        for (auto it: queries) {
            for (auto word: dictionary) {
                int count=0;
                for (int i=0; i<it.size(); i++) {
                    if (it[i]!=word[i]) count++;
                    if (count>2) break;
                }
                if (count<=2) {
                    result.push_back(it);
                    break;
                }
            }
        }
        return result;
    }
};
