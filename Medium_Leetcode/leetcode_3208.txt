class Solution {
public:
    int numberOfAlternatingGroups(vector<int>& colors, int k) {
        int n=colors.size();
        vector<int> temp=colors;
        temp.insert(temp.end(), colors.begin(), colors.end());
        int count=0;
        int i=0, j=1;
        while (j<n+k-1) {
            if (temp[j-1]==temp[j]) {
                i=j;
            }
            if (j-i+1==k) {
                i++;
                count++;
            }
            j++;
        }
        return count;
    }
};
