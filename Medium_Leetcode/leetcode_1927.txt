class Solution {
public:
    bool sumGame(string num) {
        int n=num.size();
        int leftSum=0, leftEmpty=0;
        for (int i=0; i<n/2; i++) {
            if (num[i]=='?') leftEmpty++;
            else leftSum+=(num[i]-'0');
        }
        int rightSum=0, rightEmpty=0;
        for (int i=n/2; i<n; i++) {
            if (num[i]=='?') rightEmpty++;
            else rightSum+=(num[i]-'0');
        }
        // all filled
        if (leftEmpty==0 && rightEmpty==0) {
            if (leftSum==rightSum) return false;
            return true;
        }
        // if the number of spaces is odd, alice gets the last move
        if ((rightEmpty+leftEmpty)%2==1) return true;
        else {
            return ((leftSum-rightSum)!=9*(rightEmpty-leftEmpty)/2);
        }
    }
};
