using ll=long long;
class Solution {
public:
    int countGoodRotations(vector<int>& nums) {
        int n=nums.size();
        int count=0;
        ll sum1=0, sum2=0;
        for (int i=0; i<n/2; i++) {
            sum1+=nums[i];
        }
        for (int i=n/2; i<n; i++) {
            sum2+=nums[i];
        }
        if (sum1>sum2) count++;
        for (int i=0; i<n-1; i++) {
            sum1-=nums[i]; sum1+=nums[(i+n/2)%n];
            sum2+=nums[i];  sum2-=nums[(i+n/2)%n];
            if (sum1>sum2) count++;
        }
        return count;
    }
};
