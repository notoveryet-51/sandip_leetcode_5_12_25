Leetcode 289

class Solution {
public:
    vector<int> dirRow={1, -1, 0, 0, 1, 1, -1, -1};
    vector<int> dirCol={0, 0, 1, -1, 1, -1, 1, -1};
    void gameOfLife(vector<vector<int>>& board) {
        int m=board.size();
        int n=board[0].size();
        vector<vector<int>> orig = board;
        for (int i=0; i<m; i++) {
            for (int j=0; j<n; j++) {
                int count=0;
                for (int k=0; k<8; k++) {
                    int row=i+dirRow[k];
                    int col=j+dirCol[k];
                    if (row>=0 && row<m && col>=0 && col<n && orig[row][col]==1)    count++;
                }
                if (orig[i][j]==1) {
                    if (count<2 || count>3) {
                        board[i][j]=0;
                    }
                } else {
                    if (count==3) {
                        board[i][j]=1;
                    }
                }
            }
        }
    }
};
