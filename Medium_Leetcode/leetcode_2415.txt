/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    TreeNode* reverseOddLevels(TreeNode* root) {
        if (!root) return nullptr;
        vector<vector<int>> finalTree;
        queue<TreeNode*> q;
        q.push(root);

        // Level-order traversal to populate finalTree
        while (!q.empty()) {
            int size = q.size();
            vector<int> temp;
            for (int i = 0; i < size; ++i) {
                TreeNode* currRoot = q.front();
                q.pop();
                temp.push_back(currRoot->val);
                if (currRoot->left != nullptr) q.push(currRoot->left);
                if (currRoot->right != nullptr) q.push(currRoot->right);
            }
            finalTree.push_back(temp);
        }

        // Reverse odd level vectors
        for (int level = 1; level < finalTree.size(); level += 2) { 
            reverse(finalTree[level].begin(), finalTree[level].end());
        }

        // Reassign values to the tree nodes
        q.push(root);
        int level = 0;
        while (!q.empty()) { 
            int size = q.size();
            for (int i = 0; i < size; ++i) {
                TreeNode* currRoot = q.front();
                q.pop();
                currRoot->val = finalTree[level][i];
                if (currRoot->left != nullptr) q.push(currRoot->left);
                if (currRoot->right != nullptr) q.push(currRoot->right);
            }
            level++;
        }

        return root;
    }
};
