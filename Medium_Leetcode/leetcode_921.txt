class Solution {
public:
    int minAddToMakeValid(string s) {
        int n=s.length();
        int count=0;
        int insert=0;
        for (int i=0; i<n; i++) {
            if (s[i]=='(') count++;
            else count--;
            if (count<0) {
                count=0;
                insert++;
            }
        }
        if (count>0) insert+=count;
        return insert;
    }
};
