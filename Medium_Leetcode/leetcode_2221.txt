class Solution {
    public int triangularSum(int[] nums) {
        int n=nums.length;
        int idx=n;
        if (n==1) return nums[0];
        do {
            for (int i=0; i<n-1; i++) {
                if (nums[i]!=-1 && nums[i+1]!=-1) nums[i]=(nums[i]+nums[i+1])%10;
            }
            nums[--idx]=-1;
        } while (nums[1]!=-1);
        return nums[0];
    }
}
