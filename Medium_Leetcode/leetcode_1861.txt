class Solution {
public:
    vector<vector<char>> rotateTheBox(vector<vector<char>>& boxGrid) {
        int n=boxGrid.size();
        int m=boxGrid[0].size();
        for (int i=0; i<n; i++) {
            // track the lowest available empty position for stones to fall into
            int emptyPos=m-1;
            for (int j=m-1; j>=0; j--) {
                if (boxGrid[i][j]=='*') {
                    // obstacle resets the lowest available empty slot
                    emptyPos=j-1;
                } else if (boxGrid[i][j]=='#') {
                    // move stone to the lowest available position
                    boxGrid[i][j]='.';
                    boxGrid[i][emptyPos]='#';
                    emptyPos--;
                }
            }
        }
        // rotate 90 degrees clockwise: (i, j) -> (j, n - 1 - i)
        vector<vector<char>> result(m, vector<char>(n));
        for (int i=0; i<n; i++) {
            for (int j=0; j<m; j++) {
                result[j][n-1-i]=boxGrid[i][j];
            }
        }
        return result;
    }
};
