class Solution {
public:
    int maxSubarraySumCircular(vector<int>& nums) {
        int n=nums.size();
        // wrapped sum= total sum-min subarray sum
        int sum1=0;
        int maxSum1=INT_MIN;
        int sum2=0;
        int minSum1=INT_MAX;
        int totalSum=0;
        for (int i=0; i<n; i++) {
            totalSum+=nums[i];
            sum1+=nums[i];
            maxSum1=max(maxSum1, sum1);
            if (sum1<0) sum1=0;
            
            sum2+=nums[i];
            minSum1=min(minSum1, sum2);
            if (sum2>0) sum2=0;
        }
        if (maxSum1<0) return maxSum1;
        return max(maxSum1, totalSum-minSum1);
    }
};
