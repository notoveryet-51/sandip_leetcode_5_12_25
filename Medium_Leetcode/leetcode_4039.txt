using ll=long long;
class Solution {
public:
    ll MOD=1e9+7;
    int calPower(ll x, ll y) {
        if (y==0) return 1;
        ll val=calPower(x, y/2);
        ll result=((val%MOD)*(val%MOD))%MOD;
        if (y%2!=0) return (int)(x%MOD*result%MOD)%MOD;
        return (int) result;
    }
    int sumDecoded(vector<long long>& nums) {
        ll n=nums.size();
        ll sum=0;
        for (ll i=0; i<n; i++) {
            ll width=nums[i]%10;
            string d=to_string(nums[i]/10);
            ll x, y;
            x=stoll(d.substr(0, width));
            y=stoll(d.substr(width));
            sum=(sum+calPower(x, y))%MOD;;
        }
        return (int)sum;
    }
};
