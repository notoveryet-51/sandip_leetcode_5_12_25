class Solution {
public:
    string smallestPalindrome(string s) {
        int n=s.size();
        map<char, int> freq;
        string res="";
        for (int i=0; i<n; i++) {
            freq[s[i]]++;
        }
        string ext="";
        int a;
        for (auto it: freq) {
            if (it.second%2!=0) {
                ext+=it.first; it.second--;
            }
            a=it.second/2;
            while (it.second!=a) {
                res+=it.first;
                it.second--;
            }
        }
        string next=res;
        reverse(res.begin(), res.end());
        return next+ext+res;
    }
};
