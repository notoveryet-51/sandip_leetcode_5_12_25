class Solution {
public:
    int minimumSumSubarray(vector<int>& nums, int l, int r) {
        int n=nums.size();
        int mn=INT_MAX;
        for (int i=0; i<n; i++) {
            int sum=0;
            for (int j=i; j<n; j++) {
                sum+=nums[j];
                if (sum>0 && j-i+1>=l && j-i+1<=r) mn=min(mn, sum);
            }
        }
        return (mn==INT_MAX) ? -1 : mn;
    }
};
