class Solution {
public:
    bool isPalindromic(string s) {
        string check="";
        int n=s.size();
        for (int i=0; i<n; i++) {
            int ascii=s[i];
            string temp=bitset<8>(ascii).to_string();
            check+=temp;
        }
        int len=8*n;
        for (int i=0; i<len/2; i++) {
            if (check[i]!=check[len-1-i]) return false;
        }
        return true;
    }
};
