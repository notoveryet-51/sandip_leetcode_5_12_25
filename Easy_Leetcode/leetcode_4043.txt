class Solution {
public:
    int countRotations(string s, int k) {
        int n=s.size();
        int result=0;
        int cpy=n;
        while (cpy>0) {
            int count=0;
            for (int i=0; i<n-1; i++) {
                if (s[i]==s[i+1]) count++;
            }
            if (count==k) result++;
            char c=s[0];
            s.erase(s.begin()); 
            s+=c;
            cpy--;
        }
        return result;
    }
};
