class Solution {
public:
    bool isRectangleOverlap(vector<int>& rec1, vector<int>& rec2) {
        // we use the equation approach like x-y<0 means the point is in left of the line
        // so we have 2x's here
        // we check with any rectangle...if one has point inside the other the other must have a point too since both are parallel to axes
        int x1=rec1[0], x2=rec1[2];
        int x3=rec2[0], x4=rec2[2];
        int y1=rec1[1], y2=rec1[3];
        int y3=rec2[1], y4=rec2[3];
        // check whether one rectangle is entirely on one side of the other's boundary lines
        if (x2<=x3 || x1>=x4 || y2<=y3 || y1>=y4) return false;
        return true;
    }
};
