class Solution {
public:
    void dfsHelper(int start, vector<vector<int>>& adj, bool &check, int n, vector<bool> &visited) {
        if (start==n) {
            check=true; return;
        }
        for (auto it: adj[start]) {
            if (!visited[it]) {
                visited[it]=true;
                dfsHelper(it, adj, check, n, visited);
            }
        }
    }
    bool validPath(int n, vector<vector<int>>& edges, int source, int destination) {
        vector<vector<int>> adj(n);
        for (int i=0; i<edges.size(); i++) {
            adj[edges[i][0]].push_back(edges[i][1]);
            adj[edges[i][1]].push_back(edges[i][0]);
        }
        bool check=false;
        vector<bool> visited (n, false);
        visited[source]=true;
        dfsHelper(source, adj, check, destination, visited);
        return check;
    }
};
