class Solution {
public:
    bool numberCheck(char c) {
        if (c=='1' || c=='2' || c=='3' || c=='4' || c=='5' || c=='6' || c=='7' || c=='8' || c=='9' || c=='0') return true;
        return false;
    }
    string clearDigits(string s) {
        int n=s.length();
        int i=1;
        while (i<n) {
            if (numberCheck(s[i])) {
                if (i-1>=0) s.erase(s.begin()+i-1);
                i--;
                s.erase(s.begin()+i);
                i--;
            }
            i++;
        }
        return s;
    }
};
