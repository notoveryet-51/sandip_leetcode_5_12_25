class Solution {
public:
    int minMergedInteger(vector<int>& nums) {
        int n=nums.size();
        vector<int> temp;
        for (int i=0; i<n-1; i+=2) {
            temp.push_back(stoi(to_string(nums[i])+to_string(nums[i+1])));
        }
        sort(temp.begin(), temp.end());
        return temp[0];
    }
};
