class Solution {
public:
    int countSpecialIntegers(vector<int>& nums) {
        int n=nums.size();
        vector<vector<int>> vec(101);
        for (int i=0; i<n; i++) {
            vec[nums[i]].push_back(i);
        }
        int count=0;
        for (int i=1; i<101; i++) {
            if (vec[i].empty()) continue;
            int len=vec[i].size();
            bool check=true;
            for (int j=0; j<len-1; j++) {
                if ((vec[i][j+1]-vec[i][j])!=1) {
                    check=false;
                    break;
                }
            }
            if (check) count++;
        }
        return count;
    }
};
