class Solution {
public:
    vector<vector<int>> cyclicShift(int n, vector<vector<int>>& grid, vector<int>& rowShift, vector<int>& colShift) {
        int len=grid.size();
        vector<vector<int>> temp=grid;
        for (int i=0; i<n; i++) {
            int k=((rowShift[i]%n)+n)%n;
            for (int j=0; j<n; j++) {
                temp[i][j]=grid[i][(j+k)%n];
            }
        }
        vector<vector<int>> res=temp;
        for (int j=0; j<n; j++) {
            int k=((colShift[j]%n)+n)%n;
            for (int i=0; i<n; i++) {
                res[i][j]=temp[(i+k)%n][j]; 
            }
        }
        return res;
    }
};
