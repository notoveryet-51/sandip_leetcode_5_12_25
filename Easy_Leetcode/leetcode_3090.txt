class Solution {
public:
    int maximumLengthSubstring(string s) {
        int i=0, j=0;
        unordered_map<char, int> mp;
        int maxLen=0;
        while (j<s.size()) {
            mp[s[j]]++;
            while (mp[s[j]]>2) {
                mp[s[i]]--;
                i++;
                if (mp[s[i]]==0) mp.erase(mp[s[i]]);
            }
            maxLen=max(maxLen, j-i+1);
            j++;
        }
        return maxLen;
    }
};
